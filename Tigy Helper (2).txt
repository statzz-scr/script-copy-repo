-- Function to create a toggle frame (reusable for buttons)
local function createToggleFrame(text, positionY)
    local frame = Instance.new("Frame")
    frame.Size = UDim2_New(0, 220, 0, 60)
    frame.Position = UDim2_New(0.5, -110, positionY, 0)
    frame.BackgroundColor3 = Color3_FromRGB(30, 30, 30)
    frame.BorderSizePixel = 0
    frame.AnchorPoint = Vector2_New(0.5, 0)
    frame.Active = true
    frame.Draggable = true
    frame.Parent = GameUI

    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim_New(0, 12)
    corner.Parent = frame

    local label = Instance.new("TextLabel")
    label.Size = UDim2_New(0.6, 0, 1, 0)
    label.Position = UDim2_New(0, 15, 0, 0)
    label.BackgroundTransparency = 1
    label.Text = text
    label.TextColor3 = Color3_FromRGB(255, 255, 255)
    label.Font = Enum_Font.GothamSemibold
    label.TextSize = 20
    label.TextXAlignment = Enum_TextXAlignment.Left
    label.Parent = frame

    local button = Instance.new("TextButton")
    button.Size = UDim2_New(0, 60, 0, 28)
    button.Position = UDim2_New(1, -75, 0.5, -14)
    button.BackgroundColor3 = Color3_FromRGB(50, 50, 50)
    button.BorderSizePixel = 0
    button.Text = ""
    button.Parent = frame

    local buttonCorner = Instance.new("UICorner")
    buttonCorner.CornerRadius = UDim_New(0, 14)
    buttonCorner.Parent = button

    local slider = Instance.new("Frame")
    slider.Size = UDim2_New(0.5, -2, 1, -4)
    slider.Position = UDim2_New(0, 2, 0, 2)
    slider.BackgroundColor3 = Color3_FromRGB(0, 170, 255)
    slider.BorderSizePixel = 0
    slider.Parent = button

    local sliderCorner = Instance.new("UICorner")
    slider.CornerRadius = UDim_New(0, 12)
    slider.Parent = slider

    return frame, button, slider
end

-- "No Animation" toggle
noAnimBtn.MouseButton1Click:Connect(function()
    TweenService:Create(noAnimSlider, tweenInfo, {Position = UDim2_New(0.5, -2, 0, 2)}):Play()
    
    local char = LocalPlayer.Character
    if char then
        local humanoid = char:WaitForChild("Humanoid")
        local animator = humanoid:FindFirstChildOfClass("Animator")
        if animator then
            local anim = Instance.new("Animation")
            anim.AnimationId = "rbxassetid://507766666"  -- Default idle
            local track = animator:LoadAnimation(anim)
            track.Priority = Enum.AnimationPriority.Action
            track.Looped = false
            track:Play(0)
            track.Stopped:Connect(function()
                track:Play(0)  -- Loop by replaying
            end)
        end
    end
end)

-- "Respawn Desync" toggle
desyncBtn.MouseButton1Click:Connect(function()
    TweenService:Create(desyncSlider, tweenInfo, {Position = UDim2_New(0.5, -2, 0, 2)}):Play()
    
    local char = LocalPlayer.Character
    if char then
        local humanoid = char:FindFirstChildWhichIsA("Humanoid")
        if humanoid then
            humanoid:ChangeState(Enum.HumanoidStateType.Dead)
        end
        char:ClearAllChildren()
        local fake = Instance.new("Model")
        fake.Parent = workspace
        LocalPlayer.Character = fake
        task.wait()
        LocalPlayer.Character = char
        fake:Destroy()
    end
    
    task.spawn(function()
        -- Bunch of FastFlag manipulations for desync/lag
        setfflag("GameNetPVHeaderRotationalVelocityZeroCutoffExponent", "-5000")
        setfflag("LargeReplicatorWrite5", "true")
        setfflag("LargeReplicatorEnabled9", "true")
        -- ... (many more similar flags for high bandwidth, low thresholds, etc.)
        
        -- Force jump/land and network ownership toggle
        local humanoid = LocalPlayer.Character:FindFirstChildWhichIsA("Humanoid")
        if humanoid then
            humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
            task.wait(0.1)
            humanoid:ChangeState(Enum.HumanoidStateType.Landed)
        end
        local hrp = LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        if hrp then
            pcall(function()
                hrp:SetNetworkOwner(nil)
                task.wait(0.05)
                hrp:SetNetworkOwner(LocalPlayer)
            end)
        end
    end)
end)
