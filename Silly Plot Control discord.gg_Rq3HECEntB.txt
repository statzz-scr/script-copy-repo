-- ────────────────────────────────────────────────
--                Remote Event Path
-- ────────────────────────────────────────────────

local Packages       = ReplicatedStorage:WaitForChild("Packages", 10)
local Net            = Packages and Packages:WaitForChild("Net", 5)
local PlotService    = Net and Net:WaitForChild("RE/PlotService", 5)
local ToggleFriends  = PlotService and PlotService:WaitForChild("ToggleFriends", 5)

-- Safety check — don't connect if remote doesn't exist
if not ToggleFriends or not ToggleFriends:IsA("RemoteEvent") then
    warn("[Silly's Plot Control] Could not find RemoteEvent: ReplicatedStorage.Packages.Net.RE.PlotService.ToggleFriends")
    toggleButton.Text = "Not found"
    toggleButton.BackgroundColor3 = Color3.fromRGB(80, 0, 0)
    return
end

-- ────────────────────────────────────────────────
--                Button Functionality
-- ────────────────────────────────────────────────

toggleButton.MouseButton1Click:Connect(function()
    -- Fire the remote (no arguments needed in most games that use this pattern)
    ToggleFriends:FireServer()

    -- Optional: quick visual feedback
    local originalColor = toggleButton.BackgroundColor3
    toggleButton.BackgroundColor3 = Color3.fromRGB(0, 180, 0)  -- green flash
    
    task.delay(0.4, function()
        toggleButton.BackgroundColor3 = originalColor
    end)
end)
